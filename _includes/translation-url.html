{%- comment %} Finds this page in language `include.to` and sets `translation_url`. Pages are paired by a shared `ref` in front matter, or else by having the same URL under each language's path (/about/ and /pt/about/). With no translation it falls back to that language's home and sets `translation_found` to false. {% endcomment %} {%- assign to_conf = site.languages | where: "code", include.to | first %} {%- assign to_base = to_conf.path | default: "" %} {%- assign all_docs = site.html_pages | concat: site.documents %} {%- assign translation_url = nil %} {%- if page.ref %} {%- for doc in all_docs %} {%- if doc.ref == page.ref %} {%- assign doc_lang = doc.lang | default: site.lang %} {%- if doc_lang == include.to %}{% assign translation_url = doc.url %}{% break %}{% endif %} {%- endif %} {%- endfor %} {%- endif %} {%- unless translation_url %} {%- assign rest = page.url %} {%- if lang_base != "" %}{% assign rest = page.url | slice: lang_base.size, page.url.size %}{% endif %} {%- assign candidate = to_base | append: rest %} {%- assign found = all_docs | where: "url", candidate | first %} {%- if found %}{% assign translation_url = candidate %}{% endif %} {%- endunless %} {%- assign translation_found = true %} {%- unless translation_url %} {%- assign translation_url = to_base | append: "/" %} {%- assign translation_found = false %} {%- endunless %}
