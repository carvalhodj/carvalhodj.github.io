{{- /* Sends visitors to the page in their language before anything renders. A language picked with the header flag (retro:lang) wins. Otherwise the visitor's country is guessed, first from the time zone (Brazil only), then from the browser language's region, and matched against each language's `regions` param; everyone else gets the `fallback` language (or the default). It only redirects when the page has a translation (an hreflang link above), and never for crawlers, so search engines index every language. On the 404 page the missing address decides first: /en/anything gets the English 404. The host serves one 404 for every missing URL, so the address travels along as ?path= and is put back. */ -}} {{- $opts := partial "options.html" . }} {{- $notFound := eq .Kind "404" }} {{- if and (gt (len hugo.Sites) 1) (or $opts.auto_language $notFound) }} {{- $root := (urls.Parse site.BaseURL).Path | default "/" }} {{- $prefixed := slice }} {{- $regions := slice }} {{- $fallback := partial "lang-code.html" hugo.Sites.Default.Language }} {{- range hugo.Sites }} {{- $code := partial "lang-code.html" .Language }} {{- if ne .Home.RelPermalink $root }}{{ $prefixed = $prefixed | append (dict "code" $code "home" .Home.RelPermalink) }}{{ end }} {{- with .Params.regions }}{{ $regions = $regions | append (dict "code" $code "regions" .) }}{{ end }} {{- if .Params.fallback }}{{ $fallback = $code }}{{ end }} {{- end }} {{- end }}
